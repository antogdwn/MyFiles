public class Solution {
    public int SearchInsert(int[] nums, int target) {
        //nums = [1,[?]3,5,6], target = 2
        
        int low=0; int high = nums.Length-1;
        int mid=0;
        int prevMid=(low+high)/2; //1
 
        while(low <= high ){
            mid = (low + high) /2;  //2 
            
           if(nums.Length ==1 && target < nums[low] ){
              return 0;
           }
           else if( nums.Length ==1 && target > nums[low]){
            return 1;
           }

            if(nums[mid] == target  ){ //4 ==4 true
                
                return mid;
            } 
            else if ( target > nums[mid] ){ //4 > nums[i] ?
                low = mid +1 ;
                
            }
            else{
                high = mid-1;
            }
            if(target != nums[prevMid] && target != nums[mid] && low >high && target > nums[mid]){ //2!=3 && 2!=1 && 1>0 true 1+1=2
                return mid+1;
            }
            else if(target != nums[prevMid] && target != nums[mid] && low >high && target < nums[mid]){
                return mid;
            }

            prevMid=mid;
        }
        if(low < high){
            return high;
        }
        else if(low >high){
            return low;
        }
        return 0;
    }
}
